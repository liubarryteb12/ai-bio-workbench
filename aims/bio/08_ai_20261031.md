export async function verifyPulledDeliverables({ localDeliverableDir, remoteManifestSha256, publicKeyPath }) {
  const checksumsPath = path.join(localDeliverableDir, 'CHECKSUMS.sha256');

    }
  }
  // 2. 若远端返回了 tar 包的期望 SHA256，本地拉回后必须先比对一致，严禁直接重算覆盖！
  if (remoteManifestSha256) {
