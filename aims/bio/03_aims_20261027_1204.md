### 1.4 核心算法与代码实现 3：模板选中后的真实派生执行闭环（根治缺口 G2）
修改 `BioanalysDesktop/server/figure_graph.js` 与 `server/task.js` 的 `resolvePlan`，使派生副本不再沦为“只复制不执行的摆设”，而是作为 `analysis_step` 的后置挂载任务在 K40 真实执行：

```javascript
// BioanalysDesktop/server/figure_graph.js -> buildAdaptedFigureTasks(project, line, selectedFigures)
import crypto from 'node:crypto';
import fs from 'node:fs';
import path from 'node:path';

export function buildAdaptedFigureTasks({ projectId, line, selectedFigures, templatesRegistry }) {
  const tasks = [];
  for (const sel of selectedFigures) {
    const tpl = templatesRegistry.find(t => t.id === sel.id);
    if (!tpl || tpl.deps_status !== 'ready') {
      throw new Error(`[FIGURE_GUARD] 模板 ${sel.id} 未就绪 (status=${tpl?.deps_status})，禁止派发`);
    }
    if (!tpl.lines.includes(line)) {
      throw new Error(`[FIGURE_GUARD] 模板 ${sel.id} 不支持分析线 ${line}（严禁跨线混用）`);
    }
    // 校验 adapt_vars 实参严禁包含 '..' 或其他项目/其他线路径
    for (const [k, v] of Object.entries(sel.adapt_vars || {})) {
      if (typeof v === 'string' && (v.includes('..') || /line[123]_/.test(v) && !v.includes(line))) {
        throw new Error(`[FIGURE_GUARD] adapt_vars.${k} 存在越界或跨线引用: ${v}`);
      }
    }

    const masterBuf = fs.readFileSync(tpl.source);
    const masterHash = crypto.createHash('sha256').update(masterBuf).digest('hex');
    if (masterHash !== tpl.source_sha256) {
      throw new Error(`[FIGURE_DRIFT] 母版 ${tpl.source} SHA256 漂移，请先重跑预览并更新注册表`);
    }

    // 生成项目内派生副本：local_projects/<projectId>/figures/adapt/adapt_<tpl.id>.R/.py
    const ext = tpl.language === 'r' ? '.R' : '.py';
    const localAdaptRel = path.posix.join('local_projects', projectId, 'figures', 'adapt', `adapt_${tpl.id}${ext}`);
    const header = [
      `# ====================================================================`,
      `# Master-Source: ${tpl.source}`,
      `# Master-SHA256: ${masterHash}`,
      `# Bound-Analysis-Step: ${tpl.analysis_step}`,
      `# Adapted-Vars-JSON: ${JSON.stringify(sel.adapt_vars || {})}`,
      `# ====================================================================`
    ].join('
');

    // 注入 BIO_ADAPT_VARS 环境变量读取头，驱动真实分析产物表出图
    fs.mkdirSync(path.dirname(localAdaptRel), { recursive: true });
    fs.writeFileSync(localAdaptRel, `${header}
${masterBuf.toString('utf8')}
`, 'utf8');

    // 返回真实可执行任务描述符（挂在 tpl.analysis_step 之后执行，彻底根治 G2！）
    tasks.push({
      task_id: `task_adapt_${tpl.id}`,
      depends_on_step: tpl.analysis_step,
      line,
      step: `adapt_${tpl.id}`,
      local_script: localAdaptRel,
      remote_script: `/mnt/ssd/bio_data/projects/${projectId}/scripts/adapt_${tpl.id}${ext}`,
      env_vars: {
        BIO_PROJECT_ID: projectId,
        BIO_LINE_ID: line,
        BIO_ADAPT_VARS: JSON.stringify(sel.adapt_vars || {}),
        BIO_OUTPUT_DIR: `/mnt/ssd/bio_data/projects/${projectId}/${line}/figures`
      },
      expected_outputs: tpl.produces
    });
  }
  return tasks;
}
```

### 2.1 三线生信模块（`mod-line1-bulk` / `mod-line2-scrna` / `mod-line3-spatial`）核心防崩与精度公式
1. **Line 1 独立外部集冻结验证公式（Step 9 `09_external_val.R`）**：
   $$\hat{y}_{	ext{val}, i} = \sigma\left(eta_0^{(	ext{train})} + \sum_{g \in \mathcal{H}_{	ext{core}}} eta_g^{(	ext{train})} \cdot X_{	ext{val}, i, g}ight)$$
   其中 $\mathcal{H}_{	ext{core}}$（LASSO $\cap$ RFE $\cap$ Univariate 交集 Hub 基因）与回归系数 $oldsymbol{eta}^{(	ext{train})}$ 100% 从训练集 `processed/train/frozen_model.rds` 只读加载，严禁在 `processed/val/` 上执行任何特征重选或参数重拟合。
2. **Line 2 GSVA Pseudo-bulk 降维削峰公式（Step 6 `06_gsva_pathways.R`）**：
   $$ar{E}_{g, (c, s)} = \log_2\left(rac{\sum_{k \in 	ext{Cells}(c, s)} 	ext{Count}_{g, k}}{\sum_{k \in 	ext{Cells}(c, s)} 	ext{TotalCount}_k} 	imes 10^4 + 1ight)$$
   先按“细胞亚群 $c 	imes$ 生物样本 $s$”通过 `Seurat::AggregateExpression` 聚合为伪体块矩阵 $ar{\mathbf{E}}$（列数从 $10^4$ 级细胞压降至 $10^1 \sim 10^2$ 级样本-亚群对），再输入 `GSVA::gsva()`，彻底避免 7.5GB 内存溢出。
3. **Line 2 虚拟敲除 3 种子稳健交集与 GC 削峰（Step 10 `10_virtual_ko.R`）**：
   $$	ext{DR}_{	ext{robust}} = igcap_{s \in \{20261013, 20261014, 20261015\}} \left\{ g \;\middle|\; p_{	ext{adj}}^{(s)}(g) < 0.05 \;\land\; ert Z^{(s)}(g)ert \ge 1.96 ight\}$$
   每轮 `scTenifoldKnk` 结束后立即在同一作用域内调用 `rm(res); gc(verbose = FALSE)`，确保峰值 RSS $< 4.2	ext{GB}$。
4. **Line 3 空间无监督自签名反卷积公式（Step 7 `07_nnls_deconvolution.py`）**：
   $$\hat{oldsymbol{	heta}}_{	ext{spot}} = rg\min_{oldsymbol{	heta} \ge \mathbf{0}} \left\| \mathbf{y}_{	ext{spot}} - \mathbf{S}_{	ext{spatial\_markers}} oldsymbol{	heta} ight\|_2^2, \quad 	ilde{	heta}_{	ext{spot}, k} = rac{\hat{	heta}_{	ext{spot}, k}}{\sum_{j=1}^K \hat{	heta}_{	ext{spot}, j}}$$
   签名矩阵 $\mathbf{S}_{	ext{spatial\_markers}}$ 仅由本切片各空间域 Top 5 Marker 均值构成，调用 `scipy.optimize.nnls` 逐 spot 求解并归一化，100% 遵守不跨线借用 Line 2 单细胞数据的隔离红线。
